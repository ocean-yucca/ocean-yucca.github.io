kubectl
get
# get status of nodes
kubectl get nodes

# get pods
kubectl get pods
kubectl get pod  --watch # continuously monitor pods

# get services
kubectl get services
kubectl get svc

# get deployments
kubectl get deployment
kubectl get deploy
kubectl get deployment [deploymentName] -o yaml # get configuration

# get replicasets
kubectl get replicasets
kubectl get replicaset

# get pods, services, deployments, statefulsets, replicasets
kubectl get all

# get secret
kubectl get secret

# get configMap
kubectl get configmap

# get namespace
kubectl get namespaces

# get multiple objects
kubectl get deployment,service,pod
kubectl get deploy,svc,po

# get multiple objects in a specific namespace
kubectl get deploy,svc,po -n [namespace]
create
# create pod and replicaset
# kubectl create deployment [Name] --image=image
# deployment -> replicaset -> pod -> container
kubectl create deployment redis-deploy --image=redis:alpine # create a deployment with name redis-deploy, automatically create replicaset and pod

# create component from files
# kubectl create -f [fileName]
kubectl create -f k8s/redis-deployment.yaml
# kubectl create -f [directoryName]
kubectl create -f k8s
scale
# kubectl scale deployment [deploymentName] --replicas=[num]
kubectl scale deployment webapp-deployment --replicas=3
edit
# edit deployment
# create an automatically generated compose file, edit, and save
# On save + exit, Kubernetes updates the Deployment immediately, a new ReplicaSet may be created, Pods may be restarted or replaced
# kubectl edit deployment [deploymentName]
kubectl edit deployment redis-deploy
logs
# display stdout/stderr output in containers

# one container in pod
kubectl logs [podName]

# multiple containers in pod
kubectl logs [podName] -c [containerName]
describe
# describe deployment
kubectl describe deployment [deploymentName]

# describe replicaset
kubectl describe replicaset [replicasetName]

# describe pod
kubectl describe pod [podName]

# describe service
kubectl describe service [serviceName] # get podIP:podPort connected to the service
delete
# delete deployment
kubectl delete deployment [deploymentName]
kubectl delete deploy [deploymentName]

# delete service
kubectl delete service [serviceName]
kubectl delete svc [serviceName]

# delete from configuration files
# kubectl delete -f [fileName]
kubectl delete -f k8s/redis-deployment.yaml
#kubectl delete -f [directoryName]
kubectl delete -f k8s
exec
# access container

# one container in pod
kubectl exec -it [podName] -- sh

# multiple containers in pod
kubectl exec -it [podName] -c [containerName] -- sh
apply
# create or update components
# kubectl apply -f [fileName]
kubectl apply -f k8s/redis-deployment.yaml

# kubectl apply -f [directoryName]
kubectl apply -f k8s
port-forward
# map service port to public access
# kubectl port-forward svc/[serviceName] [accessPort]:[servicePort]
kubectl port-forward svc/perceptor-service 8080:8082
api-resources
kubectl api-resources --namespaced=true # list namespaced resources
kubectl api-resources --namespaced=false # list non-namespaced resources
version
kubectl version
config
# get clusters
kubectl config get-clusters

# get current cluster
kubectl config current-context
cluster-info
# list clusters
kubectl ctx

# switch cluster
kubectl ctx [clusterName]

# display control plane and CoreDNS
kubectl cluster-info

# get debug and diagnose information
kubectl cluster-info dump
