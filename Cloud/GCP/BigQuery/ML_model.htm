ML Model
Train Model
CREATE OR REPLACE MODEL `storagedev-485602.mydataset.house_price_model`
OPTIONS(
  model_type='linear_reg',
  input_label_cols=['median_house_value'],
  data_split_method='AUTO_SPLIT'
) AS
SELECT
  longitude,
  latitude,
  housing_median_age,
  total_rooms,
  total_bedrooms,
  population,
  households,
  median_income,
  ocean_proximity,
  median_house_value
FROM `storagedev-485602.mydataset.california_housing`
WHERE median_house_value IS NOT NULL;
Evaluation Model
SELECT *
FROM ML.EVALUATE(
  MODEL `storagedev-485602.mydataset.house_price_model`
);
Use Model
SELECT
  predicted_median_house_value,
  median_income,
  total_rooms
FROM ML.PREDICT(
  MODEL `storagedev-485602.mydataset.house_price_model`,
  (
    SELECT *
    FROM `storagedev-485602.mydataset.california_housing`
    LIMIT 10
  )
);
