Ipywidgets
Load Data
from google.cloud import bigquery
import pandas as pd

client = bigquery.Client()

query = f"""
SELECT
    longitude,
    latitude,
    housing_median_age,
    total_rooms,
    total_bedrooms,
    population,
    households,
    median_income,
    ocean_proximity,
    median_house_value
FROM `{client.project}.housing.housing2`
WHERE median_house_value IS NOT NULL
"""

df = client.query(query).to_dataframe()
Layout
import ipywidgets as widgets
from IPython.display import display, clear_output
import matplotlib.pyplot as plt

# define widgets
ocean_filter = widgets.Dropdown(
    options=["All"] + sorted(df["ocean_proximity"].dropna().unique().tolist()),
    value="All",
    description="Ocean:",
    style={"description_width": "initial"}
)

income_filter = widgets.FloatRangeSlider(
    value=[
        float(df["median_income"].min()),
        float(df["median_income"].max())
    ],
    min=float(df["median_income"].min()),
    max=float(df["median_income"].max()),
    step=0.1,
    description="Income:",
    continuous_update=False,
    style={"description_width": "initial"},
    layout=widgets.Layout(width="500px")
)

age_filter = widgets.IntRangeSlider(
    value=[
        int(df["housing_median_age"].min()),
        int(df["housing_median_age"].max())
    ],
    min=int(df["housing_median_age"].min()),
    max=int(df["housing_median_age"].max()),
    step=1,
    description="House Age:",
    continuous_update=False,
    style={"description_width": "initial"},
    layout=widgets.Layout(width="500px")
)

# define output
output = widgets.Output()

def update_dashboard(change=None):

    filtered = df.copy()

    # Ocean proximity filter
    if ocean_filter.value != "All":
        filtered = filtered[
            filtered["ocean_proximity"] == ocean_filter.value
        ]

    # Income filter
    filtered = filtered[
        filtered["median_income"].between(
            income_filter.value[0],
            income_filter.value[1]
        )
    ]

    # Housing age filter
    filtered = filtered[
        filtered["housing_median_age"].between(
            age_filter.value[0],
            age_filter.value[1]
        )
    ]

    with output:
        clear_output(wait=True)

        if len(filtered) == 0:
            print("No rows match the current filters.")
            return

        # -------------------------
        # KPI section
        # -------------------------

        avg_price = filtered["median_house_value"].mean()
        avg_income = filtered["median_income"].mean()
        avg_age = filtered["housing_median_age"].mean()

        kpis = widgets.HTML(
            value=f"""
            <div style="
                display:flex;
                gap:40px;
                padding:15px;
                border:1px solid #ddd;
                border-radius:10px;
                margin-bottom:20px;
            ">
                <div>
                    <b>Homes</b><br>
                    <span style="font-size:24px">{len(filtered):,}</span>
                </div>

                <div>
                    <b>Average House Value</b><br>
                    <span style="font-size:24px">${avg_price:,.0f}</span>
                </div>

                <div>
                    <b>Average Income</b><br>
                    <span style="font-size:24px">{avg_income:.2f}</span>
                </div>

                <div>
                    <b>Average House Age</b><br>
                    <span style="font-size:24px">{avg_age:.1f}</span>
                </div>
            </div>
            """
        )

        display(kpis)

        # -------------------------
        # Three charts in ONE ROW
        # -------------------------

        fig, axes = plt.subplots(
            1,
            3,
            figsize=(18, 5),
            constrained_layout=True
        )

        # Chart 1: Income vs House Value
        axes[0].scatter(
            filtered["median_income"],
            filtered["median_house_value"],
            alpha=0.3
        )

        axes[0].set_xlabel("Median Income")
        axes[0].set_ylabel("Median House Value ($)")
        axes[0].set_title("Income vs House Value")


        # Chart 2: House Value Distribution
        axes[1].hist(
            filtered["median_house_value"],
            bins=30
        )

        axes[1].set_xlabel("Median House Value ($)")
        axes[1].set_ylabel("Number of Districts")
        axes[1].set_title("House Value Distribution")


        # Chart 3: Geographic Distribution
        scatter = axes[2].scatter(
            filtered["longitude"],
            filtered["latitude"],
            c=filtered["median_house_value"],
            alpha=0.5
        )

        axes[2].set_xlabel("Longitude")
        axes[2].set_ylabel("Latitude")
        axes[2].set_title("California Housing Prices")

        fig.colorbar(
            scatter,
            ax=axes[2],
            label="Median House Value ($)"
        )

        plt.show()
Event
ocean_filter.observe(update_dashboard, names="value")
income_filter.observe(update_dashboard, names="value")
age_filter.observe(update_dashboard, names="value")
Dashboard
controls = widgets.VBox([
    widgets.HTML("<h2>California Housing Dashboard</h2>"),
    ocean_filter,
    income_filter,
    age_filter
])

display(controls)
display(output)

update_dashboard()
