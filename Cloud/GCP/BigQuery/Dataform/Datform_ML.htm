/* definitions.housing_split.sqlx */

config {
  type: "table",
  schema: "test"
}

SELECT
  longitude,
  latitude,
  housing_median_age,
  total_rooms,
  total_bedrooms,
  population,
  households,
  median_income,
  ocean_proximity,
  median_house_value,

  MOD(
    MOD(
      FARM_FINGERPRINT(
        TO_JSON_STRING(STRUCT(longitude, latitude))
      ),
      10
    ) + 10,
    10
  ) AS split_group

FROM ${ref("housing")}

WHERE median_house_value IS NOT NULL
  AND median_income IS NOT NULL
          
/* definitions.housing_training.sqlx */

config {
  type: "table",
  schema: "test"
}

SELECT
  housing_median_age,
  total_rooms,
  total_bedrooms,
  population,
  households,
  median_income,
  ocean_proximity,
  median_house_value

FROM ${ref("housing_split")}

WHERE split_group < 8
          
/* definitions.train_housing_model.sqlx */

config {
  type: "operations",
  schema: "test",
  dependencies: ["housing_training"],
  tags: ["train_model"]
}

CREATE OR REPLACE MODEL
  `data-508912.test.housing_price_model`

OPTIONS (
  MODEL_TYPE = 'LINEAR_REG',
  INPUT_LABEL_COLS = ['median_house_value'],
  DATA_SPLIT_METHOD = 'NO_SPLIT'
)

AS
SELECT *
FROM ${ref("housing_training")};
          
/* definitions.housing_predictions.sqlx */

config {
  type: "table",
  schema: "test",
  dependencies: ["train_housing_model"],
  tags: ["prediction"]
}

SELECT
  longitude,
  latitude,
  median_income,
  ocean_proximity,
  median_house_value AS actual_price,
  predicted_median_house_value AS predicted_price,

  predicted_median_house_value
    - median_house_value AS prediction_error

FROM ML.PREDICT(
  MODEL `data-508912.test.housing_price_model`,

  (
    SELECT
      longitude,
      latitude,
      housing_median_age,
      total_rooms,
      total_bedrooms,
      population,
      households,
      median_income,
      ocean_proximity,
      median_house_value

    FROM ${ref("housing_split")}

    WHERE split_group >= 8
      AND median_income >= 8

    ORDER BY longitude, latitude
    LIMIT 20
  )
)
