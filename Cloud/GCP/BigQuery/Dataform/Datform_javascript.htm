/* includes.housing_utils.js */
// Reusable variables
const MIN_PRICE = 250000;
const MAX_PRICE = 1000000;
const MIN_INCOME = 5;

// Reusable function: classify house prices
function priceCategory(column) {
  return `
    CASE
      WHEN ${column} >= 500000 THEN 'High'
      WHEN ${column} >= 250000 THEN 'Medium'
      ELSE 'Low'
    END
  `;
}

// Reusable function: generate price filter
function priceFilter(column, min, max) {
  return `
    ${column} >= ${min}
    AND ${column} < ${max}
  `;
}

// Export variables and functions
module.exports = {
  MIN_PRICE,
  MAX_PRICE,
  MIN_INCOME,
  priceCategory,
  priceFilter
}; // makes the function available to other files.
JavaScript .js files to generate multiple tables
/* definitions.housing_by_price.js */
const categories = [
  { name: "high", min: 500000, max: 1000000 },
  { name: "medium", min: 250000, max: 500000 },
  { name: "low", min: 0, max: 250000 }
];

categories.forEach(category => {
  publish(`housing_${category.name}`, {
    type: "table"
  }).query(ctx => `
    SELECT *
    FROM ${ctx.ref("housing")}
    WHERE median_house_value >= ${category.min}
      AND median_house_value < ${category.max}
  `);
});
Built-in Javascript Functions in Dataform
  • ref(), reference a table
  • self(), reference the current output table
  • when(), conditionally generate SQL
  • incremental(), check whether the current execution uses incremental logic
  • publish(), create a table or view action in JS
  • declare(), declare an existing data source
  • assert(), define a data-quality assertion
  • operate(), define a custom SQL operation
