Dataform SQLX
Layout
  • SQLX files that define workflow actions must be located inside the definitions/ directory or its subdirectories
  • config
    • type
      • table, creates or replaces a physical table
      • view, creates or replaces a virtual table
      • incremental, creates a table initially, then updates it with new or changed data
      • declaration, registers an existing BigQuery table as a Dataform source
      • assertion, checks data quality rules and reports failures
      • operations, executes custom BigQuery SQL commands
    • database, project id
    • schema, BigQuery dataset id
    • name, table/view name
    • description, description of the action or table
    • columns, descriptions or metadata for individual output columns
    • tags, lables used to group and select Dataform actions
    • dependencies, explicit dependencies on other Dataform actions
    • disabled, prevents the action from execution
    • hermetic, whether all input dependencies are explicitly declared
    • bigquery, BigQuery-specific configuration
# workflow_setting.yaml

# project_id, specifies the GCP project for output tables and views
defaultProject: data-508912

# dataset_name, default dataset for output tables and views
defaultDataset: test 

# dataset_location, BigQuery location used to execute queries and store their results, must match the BigQuery location of both the input and output datasets
defaultLocation: US

# specify the dataset for assertion output
defaultAssertionDataset: dataform_assertions

# specifies the version of the Dataform Core framework used to compile your SQLX, SQL, and JavaScript code
# Compiling SQLX: Converting SQLX files into executable BigQuery SQL
# Managing dependencies: Understanding relationships between tables using ref()
# Managing configurations: Interpreting config { type: "table" }, schema, and other properties
# Supporting assertions: Defining data quality tests
# Generating execution order: Determining which tables must be created before others
dataformCoreVersion: "3.0.43"
        	
/* definitions.sources.california_housing.sqlx */
config {

  /* refer an existing table with Dataform without creating it */
  type: "declaration",

  /* project id for input data */
  database: "data-508912",

  /* dataset name for input data */
  schema: "test",

  /* input table name */
  name: "housing" # table_name
}

dataformCoreVersion: "3.0.43"
            
/* definitions.assertions.no_null_income.sqlx */
config {
  type: "assertion"
}

SELECT *
FROM ${ref("near_ocean")}
WHERE median_income IS NULL
Incremental
  • What to do
    • Creates a physical BigQuery table and updates it on subsequent executions instead of rebuilding the entire table
    • First execution, create a table
    • Subsequent execution, update the table
  • When to use
    • The dataset is large enough that full rebuilds are expensive or slow
    • The data changes regularly, such as hourly or daily
    • Can reliably identify new or changed records using an ID, timestamp, partition, or change-data-capture system
/* definitions.intermediate.near_ocean.sqlx */
config {
  type: "incremental",
  database: "data-508912",
  schema: "test",
  name: "near_ocean",
  description: "Incrementally load housing records near the ocean"
}

SELECT *
FROM ${ref("housing")} AS housing_source
WHERE ocean_proximity = 'NEAR OCEAN'

/*
incremental(), returns true when updating an existing incremental table
${self()}, references the current output table
*/
${when(incremental(), `
  AND NOT EXISTS (
    SELECT 1
    FROM ${self()} AS existing
    WHERE TO_JSON_STRING(existing) = TO_JSON_STRING(housing_source)
  )
`)}
Operations
  • table primarily a SQL SELECT transformation
  • operations allows custom BigQuery SQL statements such as CREATE, UPDATE, DELETE, MERGE, and ALTER
/* definitions.intermediate.operation.sqlx */
config {
  type: "operations",
  name: "create_near_ocean"
}

CREATE OR REPLACE TABLE `data-508912.test.near_ocean_backup` AS
SELECT *
FROM ${ref("housing")}
WHERE ocean_proximity = 'NEAR OCEAN';
Columns
  • Add description for columns in schema
/* definitions.intermediate.near_ocean.sqlx */
config {
  type: "table",
  database: "data-508912",
  schema: "test",
  name: "near_ocean",
  description: "Housing records with ocean_proximity equal to NEAR OCEAN",
  columns: {
    median_income: "Median income of households",
    median_house_value: "Median house value",
    ocean_proximity: "Location relative to the ocean"
  }
}
 
SELECT
  *
FROM ${ref("housing")}
WHERE ocean_proximity = "NEAR OCEAN"
Tags
  • Labels assigned to SQLX actions to organize them
  • Select and execute groups of actions by labels
/* definitions.outputs.top_100_house_value.sqlx */
config {
  type: "table",
  description: "Top 100 near-ocean housing records by median house value",
  tags: ["house_value", "output", "daily"]
}

SELECT
  *
FROM ${ref("near_ocean")}
ORDER BY median_house_value DESC
LIMIT 100
Dependencies
  • When use ${ref()} to reference other Dataform actions, automatically decide dependencies
  • Use explicit dependencies when an action must run first but is not referenced through ref()
  • The names in dependencies are the output table names defined in dependent script's config.name, not the SQLX filenames
/* definitions.outputs.top_100_house_value.sqlx */
config {
  type: "table",
  schema: "test",
  name: "top_100_house_value",

  dependencies: ["near_ocean"]
}

SELECT *
FROM `data-508912.test.near_ocean`
ORDER BY median_house_value DESC
LIMIT 100
Disabled
  • Prevents a SQLX action from executing during a workflow run
config {
  type: "table",
  database: "data-508912",
  schema: "test",
  name: "near_ocean",
  description: "Housing records near the ocean",

  disabled: true
}

SELECT *
FROM ${ref("housing")}
WHERE ocean_proximity = 'NEAR OCEAN'
Hermetic
  • true — default, all input dependencies are declared through ref() or dependencies
  • false — the action may access external tables or resources that are not registered as Dataform dependencies
config {
  type: "table",
  database: "data-508912",
  schema: "test",
  name: "near_ocean",
  hermetic: true
}

SELECT *
FROM ${ref("housing")}
WHERE ocean_proximity = 'NEAR OCEAN'
            
config {
  type: "table",
  database: "data-508912",
  schema: "test",
  name: "near_ocean",
  hermetic: false
}

SELECT *
FROM `data-508912.test.housing`
WHERE ocean_proximity = 'NEAR OCEAN'
Bigquery
  • Defines BigQuery-specific settings for the output table or view, such as partitioning, clustering, labels, and data retention
config {
  type: "table",
  database: "data-508912",
  schema: "test",
  name: "near_ocean",

  bigquery: {
    labels: { // Details -> Labels
      project: "housing",
      environment: "development",
      department: "analytics"
    }
  }
}

SELECT *
FROM ${ref("housing")}
WHERE ocean_proximity = 'NEAR OCEAN'
