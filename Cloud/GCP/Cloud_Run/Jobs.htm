Jobs
Create Image
# main.py
import time

print("Job started")
time.sleep(5)
print("Job completed")
            
FROM python:3.11-slim
WORKDIR /app
COPY main.py .
CMD ["python", "main.py"]
            
# create a repo called jobs in Artifact Repositry
gcloud builds submit --tag \
  us-central1-docker.pkg.dev/PROJECT_ID/jobs/job-demo
Create Job
gcloud run jobs create job-demo \
  --image us-central1-docker.pkg.dev/PROJECT_ID/jobs/job-demo \
  --region us-central1 \
  --tasks 1 \
  --max-retries 1
Execute Job
gcloud run jobs execute job-demo \
  --region us-central1
