Functions_framework API
Create API server with docker image
  • Pros
    • Simple
    • Be able to add files and/or models to image
  • Cons
    • One access path
Server Side
# main.py
# functions_framework runs target in main.py
import functions_framework
from flask import request, jsonify

@functions_framework.http
def hello_http(request):
    request_json = request.get_json(silent=True)

    if request_json and 'name' in request_json:
        name = request_json['name']
        return jsonify(name = name, data='Hello World!')
    else:
        return jsonify(name = None, data='Need input ...')
            
# requirements.txt
functions-framework
flask
            
# Dockerfile
FROM python:3.9-slim

# Set the working directory in the container
WORKDIR /app

# Copy the current directory contents into the container at /app
COPY . /app

# Install dependencies
RUN pip install --no-cache-dir -r requirements.txt

# Expose port 8080 for the Cloud Run service
EXPOSE 8080

# Set the environment variable for the port
ENV PORT=8080

# Define the function to run when the container starts
CMD ["functions-framework", "--target", "hello_http", "--port=8080"]
Setup
# login
gcloud auth login

# select project
gcloud auth set project [PROJECT_ID]

# add a list of registries in Artifact Registry to ~/.docker/config.json
gcloud auth configure-docker
Build and Deploy
# build and push with Cloud Build
# gcloud builds submit --tag gcr.io/[project_id]/[image_name]
gcloud builds submit --tag gcr.io/spheric-gasket-476418-h5/func_api

# deploy
gcloud run deploy myapi \
  --image=gcr.io/$GOOGLE_CLOUD_PROJECT/func_api \
  --region=us-east1 \
  --allow-unauthenticated
Client Side
import requests
 
url = "api_url"
 
# hello_http is called no matter which path is
resp = requests.post(f"{url}", json={"name": "lin", "password": "my_password"})
result = resp.json()
print(result)

resp = requests.post(f"{url}/data", json={"name": "lin", "password": "my_password"})
result = resp.json()
print(result)

result = resp.json()
