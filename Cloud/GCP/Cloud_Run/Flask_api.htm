Server Side
# app.py
from flask import Flask, request, jsonify
 
app = Flask(__name__)
 
@app.post("/data")
def get_data():
    # get username and password
    data = request.get_json()

    if data:
        username = data.get("name", "")
        password = data.get("password", "")

        return jsonify(username = username)
    else:
        return jsonify(username = None)

if __name__ == "__main__":
    app.run(host='0.0.0.0', port=8080)
            
# requirements.txt
flask
            
# Dockerfile
FROM python:3.9-slim

# Set the working directory in the container
WORKDIR /app

# Copy the current directory contents into the container at /app
COPY . /app

# Install dependencies
RUN pip install --no-cache-dir -r requirements.txt

# Expose port 8080 for the Cloud Run service
EXPOSE 8080

# Set the environment variable for the port
ENV PORT=8080

# Define the function to run when the container starts
CMD ["python", "app.py"]
Setup
# login
gcloud auth login

# select project
gcloud auth set project [PROJECT_ID]

# add a list of registries in Artifact Registry to ~/.docker/config.json
gcloud auth configure-docker
Build and Deploy
# build and push with Cloud Build
# gcloud builds submit --tag gcr.io/[project_id]/[image_name]
gcloud builds submit --tag gcr.io/spheric-gasket-476418-h5/flask_api

# deploy
gcloud run deploy myapi \
  --image=gcr.io/$GOOGLE_CLOUD_PROJECT/flask_api \
  --region=us-east1 \
  --allow-unauthenticated
