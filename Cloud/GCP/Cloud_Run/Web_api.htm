  • Pros

  • Cons

Serve Side
Cloud Run -> Services -> Deploy container -> Use an inline editor to create a function 
import functions_framework
from flask import Flask, request, jsonify

@functions_framework.http
def hello_http(request):
    request_json = request.get_json(silent=True)

    if request_json and 'name' in request_json:
        name = request_json['name']
        return jsonify(name = name, data='Hello World!')
    else:
        return jsonify(data='Need input ...')
Client Side
import requests
 
url = "api_url"
 
# hello_http is called no matter what the path is
resp = requests.post(f"{url}", json={"name": "lin", "password": "my_password"})
result = resp.json()
print(result)

resp = requests.post(f"{url}/data", json={"name": "lin", "password": "my_password"})
result = resp.json()
print(result)

resp = requests.post(f"{url}/hello_http", json={"name": "lin", "password": "my_password"})
result = resp.json()
print(result)

resp = requests.post(f"{url}/bye_htpp", json={"name": "lin", "password": "my_password"})
