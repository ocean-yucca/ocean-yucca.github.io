# view section

# 1. regular LookML view that points to an existing table
view: view_name {
	sql_table_name: `dataset_name.table_name`
}

# 2. derived table without a persistence setting
view: view_name {
	derived_table: {
		sql:
			select *
			from `dataset_name.table_name` ;;
	}
}

# 3. persistent derived table (PDT)
view: view_name {
	derived_table: {
		sql:
			select *
			from `dataset_name.table_name` ;;
	}

	datagroup_trigger: datagroup_name # defined in model
}
            
# dimension section
# dimension defines a field that users can select, group by, filter, or display

dimension: dimension_name {
	label: "field name displayed to users"
	type: type_name
	sql: ${TABLE}.field_name
}
            
# measure section
# measure defines a calculated or aggregated value in Looker

measure: measure_name {
	type: count
}

measure: measure_name {
	type: sum
	sql: ${field_name} # tells Looker which field or SQL expression to use in the measure calculation
}
