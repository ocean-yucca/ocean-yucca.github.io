import requests

def search_video(query, API_KEY):
    MAX_RESULTS = 5
    URL = "https://www.googleapis.com/youtube/v3/search"

    params = {
        "part": "snippet",
        "q": query,
        "type": "video",
        "maxResults": MAX_RESULTS,
        "key": API_KEY,
    }

    response = requests.get(URL, params=params)
    data = response.json()

    return data
            
API_KEY = 'your_key'
search_query = "swim"

search = search_video(search_query, API_KEY)

for item in search['items']:
    print('thumbnail: ', item['snippet']['thumbnails']['high'])
    print('url: ', 'https://youtu.be/'+item['id']['videoId'])
