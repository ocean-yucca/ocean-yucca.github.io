# Create
import requests
from google_auth_oauthlib.flow import InstalledAppFlow
import json

# get token for creating
SCOPES = ['https://www.googleapis.com/auth/contacts']

flow = InstalledAppFlow.from_client_secrets_file('credentials.json', SCOPES)
creds = flow.run_local_server(port=0)
# user grant permission on web browser

access_token = creds.token

# create contact
url = "https://people.googleapis.com/v1/people:createContact"

body = {
    "names": [{"givenName": "Hannah", "familyName": "Chen"}],
    "emailAddresses": [{"value": "hchen@example.com"}],
    "phoneNumbers": [{"value": "+1-757-123-4567"}],
    "organizations": [{"name": "University of Virginia", "title": "Data Scientist"}],
}

headers = {
    "Authorization": f"Bearer {access_token}",
    "Content-Type": "application/json"
}

response = requests.post(url, headers=headers, data=json.dumps(body))

# Update
import requests
from google_auth_oauthlib.flow import InstalledAppFlow
import json

# get token for updating
SCOPES = ['https://www.googleapis.com/auth/contacts']

flow = InstalledAppFlow.from_client_secrets_file('credentials.json', SCOPES)
creds = flow.run_local_server(port=0)
# user grant permission on web browser

access_token = creds.token

# get etag of target contact
RESOURCE_NAME = "people/c6347803971844442572" # each contact has a unique resource name
UPDATE_FIELDS = "emailAddresses,phoneNumbers" # update fields

get_url = f"https://people.googleapis.com/v1/{RESOURCE_NAME}"
params = {"personFields": UPDATE_FIELDS}
headers = {"Authorization": f"Bearer {access_token}"}

resp = requests.get(get_url, headers=headers, params=params)
etag = resp.json().get("etag")

# update contact with etag
patch_url = f"https://people.googleapis.com/v1/{RESOURCE_NAME}:updateContact"
params = {"updatePersonFields": "emailAddresses,phoneNumbers"}
headers = {
    "Authorization": f"Bearer {access_token}",
    "Content-Type": "application/json"
}
body = {
    "etag": resp.json().get("etag"),
    "emailAddresses": [{"value": "new.email@example.com"}],
    "phoneNumbers": [{"value": "+1-555-987-6543"}]
}

patch_resp = requests.patch(patch_url, headers=headers, params=params, json=body)

# Create
from google_auth_oauthlib.flow import InstalledAppFlow
from googleapiclient.discovery import build

# get credential
SCOPES = ["https://www.googleapis.com/auth/contacts"]
CREDENTIALS_FILE = "credentials.json"

flow = InstalledAppFlow.from_client_secrets_file(CREDENTIALS_FILE, SCOPES)
# user grant permission on web browser

creds = flow.run_local_server(port=0)

# create contact
service = build("people", "v1", credentials=creds)

new_contact = {
    "names": [{"givenName": "John", "familyName": "Doe"}],
    "emailAddresses": [{"value": "john.doe@example.com"}],
    "phoneNumbers": [{"value": "+1-555-123-4567"}],
    "organizations": [{"name": "Acme Corp", "title": "Data Scientist"}],
    "addresses": [{"streetAddress": "123 Main St", "city": "Charlottesville", "region": "VA", "postalCode": "22903"}],
}

created_contact = service.people().createContact(body=new_contact).execute()

from google_auth_oauthlib.flow import InstalledAppFlow
from googleapiclient.discovery import build

# get credential
SCOPES = ["https://www.googleapis.com/auth/contacts"]
CREDENTIALS_FILE = "credentials.json"  # from Google Cloud Console
flow = InstalledAppFlow.from_client_secrets_file(CREDENTIALS_FILE, SCOPES)
# user grant permission on web browser

creds = flow.run_local_server(port=0)

# get etag
RESOURCE_NAME = "people/c8814488620804988199" # each contact has a unique resource name

service = build("people", "v1", credentials=creds)

# retrieve etag of target contact
person = service.people().get(
    resourceName=RESOURCE_NAME,
    personFields="names,emailAddresses,phoneNumbers"
).execute()

etag = person.get("etag")

# update contact with etag
person["emailAddresses"] = [{"value": "updated.email@example.com"}]
person["phoneNumbers"] = [{"value": "+1-555-999-8888"}]

updated_person = service.people().updateContact(
    resourceName=RESOURCE_NAME,
    updatePersonFields="emailAddresses,phoneNumbers",
    body=person
).execute()
