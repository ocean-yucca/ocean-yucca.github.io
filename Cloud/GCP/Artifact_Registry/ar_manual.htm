# app.py
import dash
from dash import html
from dash.dependencies import Input, Output
 
app = dash.Dash(__name__)
server = app.server
 
num_click = 0
 
app.layout = html.Div([
    html.Button('Click', id = 'click'),
    html.Div('0', id = 'content')
        ])
 
@app.callback(
    Output('content', 'children'),
    Input('click', 'n_clicks'))
def display_hover_data(c):
    if c:
        global num_click
        num_click += 1
        return num_click
    else:
        return 0
 
if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8080, debug=True)

Setup
# login
gcloud auth login

# select project
gcloud auth set project [PROJECT_ID]

# add a list of registries in Artifact Registry to ~/.docker/config.json
gcloud auth configure-docker
Create Repo
gcloud artifacts repositories create app-repo \                                                          
  --repository-format=docker \
  --location=us-central1 \
  --description="Docker repo for my images"
Deposite Code to Cloud Storage
# compress source code in the project folder
tar -cvf myapp.tar .
 
# build a bucket on Cloud Storage
# copy compressed source code to Cloud Storage
# gsutil cp myapp.tar gs://[bucket_name]/[compressed_file_name]
gsutil cp myapp.tar gs://gcp-app-operations/myapp.tar
Build Image and Push Image to Artifact Registry
# gcloud builds submit --gcs-source-staging-dir [path_on_cloud_storage] --tag [region]-docker.pkg.dev/[project_id]/[repo_name]/[image_name]
gcloud builds submit --gcs-source-staging-dir gs://gcp-app-operations/myapp.tar --tag us-central1-docker.pkg.dev/projectdevenv/app-repo/dash-app
View Images on Artifact Registry
# list repos
gcloud artifacts repositories list --location=us-central1
 
# list images in a specific repo
gcloud artifacts docker images list us-central1-docker.pkg.dev/projectdevenv/app-repo
Deploy Image on Cloud Run
# deploy
gcloud run deploy myapp \
  --image=us-central1-docker.pkg.dev/projectdevenv/app-repo/dash-app \
  --region=us-east1 \
  --allow-unauthenticated
