# app.py
import dash
from dash import html
from dash.dependencies import Input, Output
 
app = dash.Dash(__name__)
server = app.server
 
num_click = 0
 
app.layout = html.Div([
    html.Button('Click', id = 'click'),
    html.Div('0', id = 'content')
        ])
 
@app.callback(
    Output('content', 'children'),
    Input('click', 'n_clicks'))
def display_hover_data(c):
    if c:
        global num_click
        num_click += 1
        return num_click
    else:
        return 0
 
if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8080, debug=True)

Deposite Code to Cloud Storage
# compress source code in the project folder
tar -cvf myapp.tar .
 
# build a bucket on Cloud Storage
# copy compressed source code to Cloud Storage
# gsutil cp myapp.tar gs://[bucket_name]/[compressed_file_name]
gsutil cp myapp.tar gs://app-operations-bucket/myapp.tar
Build Image and Push Image to gcr.io
# gcloud builds submit --gcs-source-staging-dir [path_on_cloud_storage] --tag gcr.io/[project_id]/[image_name]
gcloud builds submit --gcs-source-staging-dir gs://app-operations-bucket/myapp.tar --tag gcr.io/$GOOGLE_CLOUD_PROJECT/dash_app
View Images on gcr.io
gcloud container images list
Deploy Image on Cloud Run
# deploy
gcloud run deploy myapp \
  --image=gcr.io/$GOOGLE_CLOUD_PROJECT/dash_app \
  --region=us-east1 \
  --allow-unauthenticated
