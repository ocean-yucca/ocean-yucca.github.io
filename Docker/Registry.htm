Registry
  • Public registry, docker hub
  • Private registry, GitLab Registry, ECR (AWS), ACR (Azure), GCR (Google)
  • Local registry
    • Login Registry
    # login Docker Hub
docker login 
docker login https://index.docker.io/v1/

# login a private registry
docker login [domainName]
    Local Registry
    # start a local registry
docker run -d -p 8000:5000 --restart=always --name registry registry:2

# push image
docker tag redis localhost:8000/redis
docker push localhost:8000/redis

# pull image
docker pull localhost:8000/redis

# list images on local registry
http://localhost:8000/v2/_catalog
    Multiple Registries
  • Compose file contains images from multiple registries
    • version: "3.9"  # optional since v1.27.0
services:
  mysqlserver:
    image: [privateRegistry]/mysql:8 # pull from a private registry
    volumes:
      - type: volume
        source: sandbox-home
        target: /var/lib/mysql
    environment:
      - MYSQL_ROOT_PASSWORD=root
  phpmyadmin:
    image: phpmyadmin/phpmyadmin # pull from Docker Hub
    ports:
      - 8080:80
    environment:
      - PMA_HOST=mysqlserver
volumes:
  sandbox-home:
