# app.py
import dash
from dash import html
from dash.dependencies import Input, Output

app = dash.Dash(__name__)
server = app.server

num_click = 0

app.layout = html.Div([
    html.Button('Click', id = 'click'),
    html.Div('0', id = 'content')
        ])

@app.callback(
    Output('content', 'children'),
    Input('click', 'n_clicks'))
def display_hover_data(c):
    if c:
        global num_click
        num_click += 1
        return num_click
    else:
        return 0

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8080, debug=True)
Create arm64 Image
# create arm64 image on Mac
# arm64 is for Mac chips
# podman has no buildx in docker
podman build -t lchenlangley/click:arm64 .
		
# check archiecture
podman inspect Image_ID --format '{{.Architecture}}'
Push arm64 Image to Registry
# push the arm64 image to Docker Hub
podman login docker.io

podman push lchenlangley/click:arm64
Create amd64 Image
# create amd64 image on Mac
# amd64 is for x86-64 chips

# create image and push to registry without load the image to local
podman build --platform linux/amd64 -t lchenlangley/click:amd64 .
Push amd64 Image to Registry
# push the arm64 image to Docker Hub
podman push lchenlangley/click:amd64
Deployment
# pull image to local on Mac
podman pull lchenlangley/click:arm64

# podman run -p [docker_host_port]:[container_port] [image_name]
podman run -p 8080:8080 lchenlangley/click:arm64

# host_ip:host_port
http://localhost:8080
