Virtual Environment
uv
  • One project with one virtual environment
  • Create a virtual environment with a Python distribution Python downloaded by uv from Astral’s trusted Python distribution sources
# create virtual environment
# uv venv .[venv_name]
uv venv # default name is .venv
uv venv .venv

# create virtual environment with specific python version
# uv venv .[venv_name] --python [version]
uv venv .venv --python 3.10.19

# activate virtual environment
# source .[venv_name]/bin/activate
source .venv/bin/activate

# deactivate
deactivate

# remove virtual environment
# rm -rf .[venv_name]
rm -rf .venv
poetry
  • One project with one virtual environment
# install python3.11 and python3.13 with pyenv

# create virtual environment if not exist
# activate virtual environment if exist
poetry env use python3.11

# list virtual environment
poetry env list

# change virtual environment
# create virtual environment if not exist
# activate virtual environment if exist
poetry env use python3.13

# check python version
poetry run python --version

# information of virtual environment
poetry env info

# no need to deactivate

# remove a virtual environment
poetry env remove python3.11
conda
  • One virtual environment for multiple projects
  • Create a virtual environment with a Conda own Python distribution
# create virtual environment with specific python version
conda create -n myenv python=3.11

# list conda virtual environments
conda env list

# activate a virtual environment
# conda activate [venv_name]
conda activate myenv

# deactivate
conda deactivate

# remove a virtual environment
# conda remove -n [venv_name] --all
conda remove -n myenv --all
virtualenv
  • One project with one virtual environment
  • Create a virtual environment with the system Python version or the version specified by pyenv
# create virtual environment
# virtualenv .[venv_name]
virtualenv .venv # use system python version
python -m virtualenv .venv # use the python version specified by pyenv

# create virtual environment with a specific python version
pyenv install 3.13.11
pyenv local 3.13.11
python -m virtualenv .venv

# activate virtual environment
# source .[venv_name]/bin/activate
source .venv/bin/activate

# deactivate
deactivate

# remove virtual environment
# rm -rf .[venv_name]
rm -rf .venv
venv
  • One project with one virtual environment
  • Create a virtual environment with the current running Python version
# create virtual environment
# python -m venv .[venv_name]
python -m venv .vname

# create virtual environment with specific python version
pyenv install 3.13.11
pyenv local 3.13.11
python -m venv .vname

# activate virtual environment
# source .[venv_name]/bin/activate
source .vname/bin/activate

# deactivate
deactivate

# remove virtual environment
# rm -rf .[venv_name]
rm -rf .vname
