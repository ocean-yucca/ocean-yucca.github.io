# app.py
import os
from flask import Flask, render_template

app = Flask(__name__)

@app.route('/')
@app.route('/page_1')
def get_page_1():
    return render_template('page_1.html')

@app.route('/page_2')
def get_page_2():
    return render_template('page_2.html')

if __name__ == '__main__':
    app.run(port = 8080)
		
# page_1.html
<!doctype html>
<html>
    <body>
        <h1>Page 1</h1>
        <a href = "{{url_for('get_page_2')}}">Go to Page 2</a>
    </body>
</html>

# app.py
import os
from flask import Flask, render_template

app = Flask(__name__)

@app.route('/')
def get_page():
    return render_template('page.html', items = ['a', 10, 'b'])

if __name__ == '__main__':
    app.run(port = 8080)
		
# page.html
<!doctype html>
<html>
    <body>
        {% for item in items %}
            <li>{{item}}</li>
        {% endfor %}
    </body>
</html>
