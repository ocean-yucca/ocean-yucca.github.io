  • Variable name in route must be same as the parameter name in function

from flask import Flask

app = Flask(__name__)

@app.route('/route/<var>')
def get_var(var):
    print(var, type(var)) # 10, str, default
    return "Get variable: "+str(var)

    import os
from flask import Flask, url_for, request

app = Flask(__name__)

@app.route('/search')
def get_search():
    name = request.args.get('name')
    id_num = request.args.get('id')

    url = url_for('get_search', name=name, id=id_num) # /search?name=lin&id=10

    return url

if __name__ == '__main__':
    app.run(port = 8080)

# http://localhost:8080/search?name=lin&id=10
# curl "http://localhost:8080/search?name=lin&id=10"

from flask import Flask, request, jsonify, Response

app = Flask(__name__)

@app.get("/ping") # equals to @app.route("/ping", methods=["GET"])
def ping():
    # read query parameters from url
    print('parameters: ', request.args)

    # inspect headers
    print('headers:', request.headers)

    return Response(
            response = jsonify({'Name: ': 'Lin', 'Age: ': 25}).get_data(as_text=True), # body
            status = 200, # status
            mimetype="application/json", # content type
            headers={ # headers
            "X-Custom-Header": "DemoValue",
            "Cache-Control": "no-cache"
        })

if __name__ == "__main__":
    app.run(port=8080)

# http://localhost:8080/ping?name=lin&n=10
# curl "http://localhost:8080/ping?name=lin&n=10"

    # server.py
from flask import Flask, request, jsonify

app = Flask(__name__)

@app.post("/submit") # equals to @app.route("/submit", methods=["POST"])
def submit():
    # client sends JSON
    json_data = request.get_json(silent=True) or {}

    # client sends data
    form_data = request.form.to_dict()

    return jsonify({
        "received_json": json_data,
        "received_form": form_data
    }), 200

if __name__ == "__main__":
    app.run(port=8080)

    Upload

    # app.py
import os
from flask import Flask, request, render_template
from werkzeug.utils import secure_filename

app = Flask(__name__)
UPLOAD_FOLDER = 'uploads'
app.config['UPLOAD_FOLDER'] = UPLOAD_FOLDER

@app.route('/')
def get_input():
    return render_template('form.html')

@app.route('/upload', methods=['POST'])
def upload_file():
    uploaded_file = request.files.get('file')

    if uploaded_file and uploaded_file.filename != '':
        filename = secure_filename(uploaded_file.filename) # avoid dangerous paths
        file_path = os.path.join(app.config['UPLOAD_FOLDER'], filename)
        uploaded_file.save(file_path)
        return f"File saved to {file_path}"

    return "No file uploaded"

if __name__ == '__main__':
    app.run(port = 8080)
        
    # form.html
<!doctype html>
<html>

        <form action="/upload" method="post" enctype="multipart/form-data">
            <input type="file" name="file">
            <input type="submit" value = "Upload">
        </form>
    </body>
</html>

    # server.py
from flask import Flask, request, jsonify, Response, send_file

app = Flask(__name__)

@app.get("/string")
def get_string():
    return "Hello, world"

@app.get('/tuple')
def get_tuple():
    return "Hello", 201, {"X-Custom": "Value"} # body, status, headers

@app.get('/dict')
def get_dict():
    return {"message": "hello"} # json

@app.get("/json")
def get_json():
    return jsonify(message="hello", score=99) # json

@app.get("/response")
def get_resp():
    resp = Response(response="Hello", status=200, mimetype="text/plain")
    resp.headers["X-Custom"] = "Value"
    return resp

@app.get("/file")
def get_file():
    return send_file("readme.md", as_attachment=True) # download

if __name__ == "__main__":
    app.run(port=8080)

# http://localhost:8080/string

    Stream Output
    # server.py
from flask import Flask, Response
import time

app = Flask(__name__)

def generate_numbers():
    for i in range(5):
        yield f"Number: {i}\n"  # Send a chunk
        time.sleep(1)           # Wait before sending next chunk

@app.get("/stream")
def stream():
    return Response(generate_numbers(), mimetype="text/plain")

if __name__ == "__main__":
    app.run(port=8080)
        
    # client.py
import requests

url = "http://127.0.0.1:8080/stream"

with requests.get(url, stream=True, timeout=10) as r:
    r.raise_for_status() # raises an exception if the server returned an error status
    for line in r.iter_lines(decode_unicode=True):
        if line:
            print("got:", line)
