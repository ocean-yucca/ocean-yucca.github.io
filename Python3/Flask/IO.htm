Parameter in Route
  • Pass value to prameter in function via route
  • Variable name in route must be same as the parameter name in function
    • import os
from flask import Flask

app = Flask(__name__)

@app.route('/route/<var>')
def get_var(var):
    print(var, type(var)) # 10, str, default
    return "Get variable: "+str(var)

if __name__ == '__main__':
    app.run(port = 8080)

# http://localhost:8080/route/10
# pass 10 to var of get_var()

    import os
from flask import Flask, url_for, request

app = Flask(__name__)

@app.route('/search')
def get_search():
    name = request.args.get('name')
    id_num = request.args.get('id')

    url = url_for('get_search', name=name, id=id_num) # /search?name=lin&id=10

    return url

if __name__ == '__main__':
    app.run(port = 8080)

# http://localhost:8080/search?name=lin&id=10
# curl "http://localhost:8080/search?name=lin&id=10"

    Upload

    # app.py
import os
from flask import Flask, request, render_template
from werkzeug.utils import secure_filename

app = Flask(__name__)
UPLOAD_FOLDER = 'uploads'
app.config['UPLOAD_FOLDER'] = UPLOAD_FOLDER

@app.route('/')
def get_input():
    return render_template('form.html')

@app.route('/upload', methods=['POST'])
def upload_file():
    uploaded_file = request.files.get('file')

    if uploaded_file and uploaded_file.filename != '':
        filename = secure_filename(uploaded_file.filename) # avoid dangerous paths
        file_path = os.path.join(app.config['UPLOAD_FOLDER'], filename)
        uploaded_file.save(file_path)
        return f"File saved to {file_path}"

    return "No file uploaded"

if __name__ == '__main__':
    app.run(port = 8080)
        
    # form.html
<!doctype html>
<html>
    <body>
        <form action="/upload" method="post" enctype="multipart/form-data">
            <input type="file" name="file">
            <input type="submit" value = "Upload">
        </form>
    </body>
</html>
