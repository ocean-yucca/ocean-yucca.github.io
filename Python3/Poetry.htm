pyproject.toml 
[build-system]
requires = ["poetry-core>=1.0.0"]
build-backend = "poetry.core.masonry.api"

# Project Metadata
[tool.poetry]
name = "primepackage"
version = "0.1.0"
description = "Demo project managed by Poetry"
authors = ["firstname lastname <firstname.lastname@example.com>"]
readme = "README.md"

packages = [
  { include = "primepackage", from = "src" }
]

# Include Extra Files
include = [
  { path = "README.md", format = "sdist" }, # ignorde
  { path = "LICENSE", format = "sdist" }, # ignored
  { path = "data/*.csv", format = "sdist" } # will be copied to the package folder
]

# List of Dependencies
[tool.poetry.dependencies]
python = "^3.10"
requests = "^2.31"
pandas = "^2.0"
matplotlib = "^3.10.6"

# Executable Script
[tool.poetry.scripts]
myprimes = "primepackage.generator:main"

# Dev extras
[tool.poetry.group.dev.dependencies]
pytest = "^7.0"
black = "^23.0"

# Unit test
[tool.pytest.ini_options]
testpaths = ["tests"]
addopts = "-ra"

Development
  • poetry install
    • Create a virtual environment using pyproject.toml for the current project
    • Install dependencies
    • Create poetry.lock
    • Install the project package and executable script in the virtual environment in editable mode, equals to pip install -e .
    • poetry install
    display virtual environment 
    poetry env info
    execute script in virtual environment 
    # run executable script
poetry run myprimes
		
    # run interactive python
poetry run python
		
    # run a script which uses the project package

# test.py
import primepackage

print(primepackage.is_prime(13))

poetry run python test.py
    add new dependency 
    poetry add matplotlib
    implement unit test 
    poetry run pytest
