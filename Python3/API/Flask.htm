# server.py
from flask import Flask, request, jsonify
import secrets
import time

app = Flask(__name__)

@app.post("/token")
def get_token():
    # get username and password
    data = request.get_json(silent=True) or request.form or {}
    username = data.get("username", "")
    password = data.get("password", "")
    print(username, password)

    # validate username and password

    # send token
    token = secrets.token_urlsafe(32)
    return jsonify(access_token=token, token_type="Bearer"), 200

@app.get("/data")
def get_data():
    # get token
    token = request.headers.get("Authorization", "")
    print(token)

    # validate token

    # return data
    return jsonify(data='Hello World!'), 200

if __name__ == "__main__":
    app.run(port=8080)

API Methods
GET
  • Retrieve data from the server
  • Read only
    • # server.py
from flask import Flask, request, jsonify, Response
import pandas as pd

app = Flask(__name__)
df = pd.read_csv('housing.csv')
df = df.where(pd.notnull(df), None) # Clean NaNs

@app.get("/data")
def get_data():
    json_str = df.to_json(orient="records")
    return Response(json_str, mimetype="application/json")

@app.get("/data/<int:start>/<int:end>")
def get_rows(start, end):
    json_str = df.iloc[start:end, :].to_json(orient="records")
    return Response(json_str, mimetype="application/json")

@app.get("/data/size")
def get_data_size():
    size = len(df)
    return jsonify(size = size)

if __name__ == "__main__":
    app.run(port=8080)

    POST
  • Retrieve data from the server
  • Send data to the server to create a resource or trigger an action
  • Change server state
    • # server.py
from flask import Flask, request, jsonify, Response
import pandas as pd

app = Flask(__name__)
df = pd.read_csv('housing.csv')
df = df.where(pd.notnull(df), None) # Clean NaNs

@app.post("/data")
def get_data():
    data = request.get_json(silent=True)
    start = data['start']
    end = data['end']

    json_str = df.iloc[start:end, :].to_json(orient="records")
    return Response(json_str, mimetype="application/json")

if __name__ == "__main__":
    app.run(port=8080)

    PUT
  • Create or fully replace a resource
    • # server.py
from flask import Flask, request, jsonify, Response
import pandas as pd

app = Flask(__name__)
df = pd.read_csv('housing.csv')
df = df.where(pd.notnull(df), None) # Clean NaNs

@app.put("/insert")
def insert_row():
    global df
    new_row = request.get_json(silent=True)
    print(new_row)

    size_before = len(df)

    df = pd.concat([df, pd.DataFrame([new_row])], ignore_index=True)
    size_after = len(df)

    print(size_before, size_after)

    return jsonify(before=size_before, after=size_after)

if __name__ == "__main__":
    app.run(port=8080)

    DELETE
  • Remove a resource
    • # server.py
from flask import Flask, request, jsonify, Response
import pandas as pd

app = Flask(__name__)
df = pd.read_csv('housing.csv')
df = df.where(pd.notnull(df), None) # Clean NaNs

@app.delete("/delete/<int:index>")
def delete_row(index):
    global df
    size_before = len(df)

    df = df.drop(index=index)
    size_after = len(df)

    print(size_before, size_after)

    return jsonify(before=size_before, after=size_after)

if __name__ == "__main__":
    app.run(port=8080)

    Status
  • 1xx: informational responses
    • Let the client know the server has received the request and is continuing to process it
    • 102 Processing
  • 2xx: Success
    • The client’s request was received, understood, and accepted
    • 200 OK, 201 Created, 202 Accepted
  • 3xx: Redirection
    • The resource isn’t here directly, use a different URL
    • 300 Multiple Choices, 301 Moved Permanently
  • 4xx: Client Error
    • Something is wrong with the request (syntax, authentication, permissions, etc.)
    • 400 Bad Request, 401 Unauthorized, 403 Forbidden, 404 Not Found
  • 5xx: Server Error
    • Request was valid, but the server failed to fulfill it
    • 500 Internal Server Error
