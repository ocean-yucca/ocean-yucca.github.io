from fastapi import FastAPI, Body, Header
from fastapi.responses import JSONResponse
import secrets

app = FastAPI()

@app.post("/login")
async def login(username:str = Body(...), password:str = Body(...)):
    print(username, password)
    # validate username and password

    # send token
    token = secrets.token_urlsafe(32)
    return JSONResponse({'access_token':token, 'token_type':"Bearer"})

@app.get("/data")
async def get_data(token: str = Header(...)):
    print(token)

    # validate token

    return JSONResponse({'data': 'content'})

from fastapi import FastAPI, Response
import pandas as pd

app = FastAPI()
df = pd.read_csv('housing.csv')
df = df.where(pd.notnull(df), None) # Clean NaNs

@app.get("/data")
async def get_data():
    json_str = df.to_json(orient="records")
    return Response(content=json_str, media_type="application/json")

@app.get("/data/{start}/{end}")
async def get_rows(start:int, end:int):
    json_str = df.iloc[start:end, :].to_json(orient="records")
    return Response(json_str, media_type="application/json")

@app.get("/data/size")
async def get_data_size():
    size = len(df)
    return {'size': size}

from fastapi import FastAPI, Body, Response
from pydantic import BaseModel
import pandas as pd

app = FastAPI()
df = pd.read_csv('housing.csv')
df = df.where(pd.notnull(df), None) # Clean NaNs

class Item(BaseModel):
    longitude: float
    latitude: float
    housing_median_age: float
    total_rooms: float
    total_bedrooms: float
    population: float
    households: float
    median_income: float
    median_house_value: float
    ocean_proximity: str

@app.put("/insert")
async def insert_rows(row: Item):
    global df
    size_before = len(df)

    df = pd.concat([df, pd.DataFrame([row.dict()])], ignore_index=True)

    size_after = len(df)
    return {'Size before insertion: ', size_before, 'Size after insertion: ', size_after}

@app.post("/upload")
async def upload_file(file: UploadFile = File(...)):
    # save the received file
    with open('housing2.csv', "wb") as out:
        while True:
            chunk = await file.read(1024 * 1024)  # 1 MB
            if not chunk:
                break
            out.write(chunk)

    return {"filename": file.filename}

    # server.py
from fastapi import FastAPI
from fastapi.responses import Response, JSONResponse, HTMLResponse, PlainTextResponse, FileResponse
import json

app = FastAPI()

@app.get("/string")
async def get_string():
    return PlainTextResponse('Hello World!')

@app.get("/html")
async def get_html():
    return HTMLResponse('<h1>Hello</h1>')

@app.get("/json")
async def get_json():
    return JSONResponse({'username': 'Lin'})

@app.get("/file")
async def get_file():
    return FileResponse("./housing.csv", media_type="application/csv")

@app.get("/raw_string")
def get_raw_string():
    return Response(content="Hello World!", media_type="text/custom")

@app.get("/raw_json")
def get_raw_json():
    data = {"ok": True, "items": [1, 2, 3]}
    return Response(content=json.dumps(data, ensure_ascii=False), media_type="application/json", status_code=200)
