innerHTML
  • Reads/writes HTML
    • <div id = "demo"></div>

<script>
	document.getElementById("demo").innerHTML = '<b style="color:red">Hello Javascript</b>';
</script>
    textContent
  • Reads/writes plain text
    • <div id = "demo2"></div>
    <script>
	    document.getElementById("demo2").textContent = "Hello World!";
    </script>
    document.write
  • Write text, HTML, a variable, or a calculated value
  • Using document.write() after an HTML document is loaded, will delete all existing HTML
    • <script>
	document.write(5+6)
</script>
    window.alert
  • Shows a simple popup message
    • <script>
	window.alert(5 + 6);
</script>
    console.log
  • Prints information to the browser's developer console
    • <script>
	console.log("Hello World!");
	console.log('%s', 'hello');   // string
	console.log('%d', 123);       // integer
	console.log('%f', 3.14); // float
</script>
    console.table
  • Displays object/array data as a table
    • const cars = [
  { brand: "Toyota", year: 2025 },
  { brand: "Honda", year: 2024 }
];

console.table(cars);
