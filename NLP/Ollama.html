Ollama
Local run
# load model to .llama/models and start a server
ollama run phi3

# /?, help

# display the process of Ollama server
ps aux | grep ollama

# displary the port of Ollama server
lsof -i -P -n | grep ollama # by default, 11434
Commands
# ollama pull [model], pull a model from Ollma library to .llama/models
ollama pull phi3

# ollama run [model], run and chat with a modle
ollama run phi3

# ollama rm [model], remove a model from local server
ollama rm phi3

# ollama list, list models on local server
ollama list

# ollama cp [model_1] [model_2], copy a model
ollama cp phi3 my_model

# ollama show [model], show model information
ollama show phi3
Python API
import ollama

# install model
ollama.pull('phi3')

# delete model
ollama.delete('phi3')

# list models
ollama.list()

# show model information
ollama.show('gemma3')

# create model with customized system instruction and parameters
ollama.create(
    model="ds-tutor",
    from_="llama3.2",
    system="You are a helpful data science tutor. Explain concepts clearly and give Python examples when useful.Keep answers concise.",
    parameters={
        "temperature": 0.3,
        "num_ctx": 8192
    }
)

# create a copy of model
ollama.copy('phi3', 'my_model')

# push a model to Ollama
ollama.copy('phi3', 'user_name/model_name') # create a copy with user_name/model_name
# ollama signin # login in the command line
# create a model with the name of model_name
ollama.push('user_name/my_model')
		
from ollama import Client
client = Client(host='http://localhost:11434')

# one-turn interaction
client.generate(model='my_model', prompt='What is the capital of France?') # The capital of France is Paris

# Multi-Turn Interaction
messages = [
    {"role": "system", "content": "You are a helpful tutor."},

    {"role": "user", "content": "What is the capital of France?"},
    {"role": "assistant", "content": "The capital of France is Paris."},

    {"role": "user", "content": "What country is it in?"},
    {"role": "assistant", "content": "Paris is in France."},

    {"role": "user", "content": "What language do people speak there?"}
]

response = client.chat(
    model="my_model",
    messages=messages
)

# roles
# system, instructions for how the assistant should behave
# user, the human’s message/question
# assistant, the model’s previous answer
# tool, result from an external tool/function
messages = []

# round 1
messages.append({
    "role": "user",
    "content": "What is the capital of France?"
})

response = client.chat(
    model="my_model",
    messages=messages
)

messages.append({
    "role": "assistant",
    "content": response["message"]["content"]
})

# round 2
messages.append({
    "role": "user",
    "content": "What country is it in?"
})

response = client.chat(
    model="my_model",
    messages=messages
)

messages.append({
    "role": "assistant",
    "content": response["message"]["content"]
})

# streaming
stream = ollama.chat(
    model='phi3',
    messages=[{'role': 'user', 'content': 'Why is the sky blue?'}],
    stream=True,
)

for chunk in stream:
  print(chunk['message']['content'], end='', flush=True)
		
# cloud model
# ollama does not download the full model weights to local when use cloud model

response = ollama.chat(
    model='gpt-oss:120b-cloud',
    messages=[{'role': 'user', 'content': 'Hello!'}],
)
print(response.message.content)
		
# embedding
