# runs and orchestrates sub-agents
import type { ExtensionAPI } from "@earendil-works/pi-coding-agent";
import { Type } from "typebox";
import { spawn } from "node:child_process";
import { findAgent, type AgentScope } from "./agents";

function runCommand(
  command: string,
  args: string[],
  cwd: string,
  input: string
): Promise<{ code: number | null; stdout: string; stderr: string }> {
  return new Promise((resolve) => {
    const child = spawn(command, args, {
      cwd,
      stdio: ["pipe", "pipe", "pipe"],
    });

    let stdout = "";
    let stderr = "";

    child.stdout.on("data", (chunk) => {
      stdout += chunk.toString();
    });

    child.stderr.on("data", (chunk) => {
      stderr += chunk.toString();
    });

    child.on("close", (code) => {
      resolve({ code, stdout, stderr });
    });

    child.stdin.write(input);
    child.stdin.end();
  });
}

export default function (pi: ExtensionAPI) {
  pi.registerTool({
    name: "subagent",
    label: "Subagent",
    description:
      "Delegate one task to a specialized sub-agent with isolated context. This minimal version supports single mode only.",

    parameters: Type.Object({
      agent: Type.String({
        description: "Name of the agent to use, for example reviewer",
      }),

      task: Type.String({
        description: "Task for the sub-agent to complete",
      }),

      agentScope: Type.Optional(
        Type.Union([
          Type.Literal("user"),
          Type.Literal("project"),
          Type.Literal("both"),
        ])
      ),
    }),

    async execute(_toolCallId, params) {
      const cwd = process.cwd();
      const scope = (params.agentScope ?? "project") as AgentScope;

      const agent = await findAgent(params.agent, scope, cwd);

      if (!agent) {
        return {
          content: [
            {
              type: "text",
              text: `Sub-agent not found: ${params.agent}. Checked scope: ${scope}.`,
            },
          ],
          details: {
            agent: params.agent,
            scope,
          },
        };
      }

      const allowedTools =
        agent.tools && agent.tools.length > 0
          ? ["--tools", agent.tools.join(",")]
          : [];

      const modelArgs = agent.model ? ["--model", agent.model] : [];

      const prompt = `
${agent.prompt}

# Task

${params.task}

# Output format

Return only your final findings. Do not modify files unless your agent instructions explicitly allow it.
`.trim();

      const result = await runCommand(
        "pi",
        [
          "--mode",
          "json",
          "-p",
          "--no-session",
          ...allowedTools,
          ...modelArgs,
        ],
        cwd,
        prompt
      );

      if (result.code !== 0) {
        return {
          content: [
            {
              type: "text",
              text: `Sub-agent failed.\n\nSTDERR:\n${result.stderr}\n\nSTDOUT:\n${result.stdout}`,
            },
          ],
          details: {
            exitCode: result.code,
            stderr: result.stderr,
          },
        };
      }

      return {
        content: [
          {
            type: "text",
            text: result.stdout.trim(),
          },
        ],
        details: {
          agent: agent.name,
          scope,
          filePath: agent.filePath,
        },
      };
    },
  });

  pi.registerCommand("list-agents", {
    description: "List available sub-agents",

    handler: async (_args, ctx) => {
      const cwd = process.cwd();
      const agents = await import("./agents").then((module) =>
        module.loadAgents("both", cwd)
      );

      if (agents.length === 0) {
        ctx.ui.notify("No sub-agents found.", "info");
        return;
      }

      const text = agents
        .map(
          (agent) =>
            `${agent.name} — ${agent.description}\n  file: ${agent.filePath}`
        )
        .join("\n\n");

      ctx.ui.notify(text, "info");
    },
  });
}
        
# ~/.pi/extensions/subagent/agents.ts
# finds and loads agent definitions
import { existsSync } from "node:fs";
import { readdir, readFile } from "node:fs/promises";
import path from "node:path";

export type AgentScope = "user" | "project" | "both";

export type AgentDefinition = {
  name: string;
  description: string;
  tools?: string[];
  model?: string;
  prompt: string;
  filePath: string;
};

function parseFrontmatter(content: string): {
  metadata: Record<string, string>;
  body: string;
} {
  if (!content.startsWith("---")) {
    return {
      metadata: {},
      body: content.trim(),
    };
  }

  const end = content.indexOf("\n---", 3);

  if (end === -1) {
    return {
      metadata: {},
      body: content.trim(),
    };
  }

  const rawMetadata = content.slice(3, end).trim();
  const body = content.slice(end + 4).trim();

  const metadata: Record<string, string> = {};

  for (const line of rawMetadata.split("\n")) {
    const index = line.indexOf(":");

    if (index === -1) continue;

    const key = line.slice(0, index).trim();
    const value = line.slice(index + 1).trim();

    metadata[key] = value;
  }

  return { metadata, body };
}

function parseList(value?: string): string[] | undefined {
  if (!value) return undefined;

  return value
    .split(",")
    .map((item) => item.trim())
    .filter(Boolean);
}

async function loadAgentsFromDirectory(
  directory: string
): Promise<AgentDefinition[]> {
  if (!existsSync(directory)) {
    return [];
  }

  const files = await readdir(directory);
  const markdownFiles = files.filter((file) => file.endsWith(".md"));

  const agents: AgentDefinition[] = [];

  for (const file of markdownFiles) {
    const filePath = path.join(directory, file);
    const content = await readFile(filePath, "utf8");
    const { metadata, body } = parseFrontmatter(content);

    if (!metadata.name || !metadata.description) {
      continue;
    }

    agents.push({
      name: metadata.name,
      description: metadata.description,
      tools: parseList(metadata.tools),
      model: metadata.model,
      prompt: body,
      filePath,
    });
  }

  return agents;
}

export async function loadAgents(
  scope: AgentScope,
  projectRoot: string
): Promise<AgentDefinition[]> {
  const userAgentsDir = path.join(
    process.env.HOME ?? "",
    ".pi",
    "agent",
    "agents"
  );

  const projectAgentsDir = path.join(projectRoot, ".pi", "agents");

  const agents: AgentDefinition[] = [];

  if (scope === "user" || scope === "both") {
    agents.push(...(await loadAgentsFromDirectory(userAgentsDir)));
  }

  if (scope === "project" || scope === "both") {
    agents.push(...(await loadAgentsFromDirectory(projectAgentsDir)));
  }

  return agents;
}

export async function findAgent(
  name: string,
  scope: AgentScope,
  projectRoot: string
): Promise<AgentDefinition | undefined> {
  const agents = await loadAgents(scope, projectRoot);

  return agents.find((agent) => agent.name === name);
}
