Command
claude
# start a claude session
claude

# query without enter interactive session
claude -p "query"

# process piped content
cat readme.md | claude -p "explain"

# list previous sessions
# session history is stored in ~/.claude/projects/[project]/[session-id].jsonl
claude --resume # command line
/resume # within claude session

# resume a specific session
claude -r "[session-id]"

# continue most recent conversation in current directory
claude -c

# sign in Anthropic account
claude auth login

# log out from Anthropic account
claude auth logout

# manage plugin
claude plugin

# background session
claude agents # list the running sessions in background, use ↑ and ↓ to select session, and use → to select and ← to unselect session
claude attach [session-id] # open a session running in background
claude logs [session-id] # show recent output
claude stop [session-id] # stop session running in background
